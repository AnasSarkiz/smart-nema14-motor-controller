import { fanoutTracePath } from "@tscircuit/props"
import savedRoutes from "./saved-paths.json"

const paths = savedRoutes.paths.map((path) => fanoutTracePath.parse(path))

export function savedRoutingPhaseIndex(netName: string, probeNet?: string) {
  if (netName === probeNet) return 1
  return savedRoutes.net_names.includes(netName) ? 0 : undefined
}

/** Retain complete physical nets; unfinished nets remain for native routing. */
export function SavedRoutes({ probeNet }: { probeNet?: string }) {
  return (
    <>
      <autoroutingphase
        name="Saved local copper"
        phaseIndex={0}
        autorouter="auto_local"
        connections={savedRoutes.connections}
        pcbTracePaths={paths}
      />
      {probeNet && (
        <autoroutingphase
          name={`Route remaining ${probeNet}`}
          phaseIndex={1}
          autorouter="auto_local"
          connection={`net.${probeNet}`}
        />
      )}
    </>
  )
}
